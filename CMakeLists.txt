cmake_minimum_required(VERSION 3.16)
project(aicmd VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8 /permissive-)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
find_package(Threads REQUIRED)
add_library(aicmd_core src/core/core.cpp src/core/location.cpp src/fs/local.cpp src/fs/ssh_config.cpp src/fs/ssh_process.cpp src/fs/ssh.cpp)
target_include_directories(aicmd_core PUBLIC src)
target_link_libraries(aicmd_core PUBLIC Threads::Threads)
add_library(aicmd_state src/host/state.cpp)
target_include_directories(aicmd_state PRIVATE third_party)
target_link_libraries(aicmd_state PUBLIC aicmd_core)
add_library(aiview_core src/media/session.cpp)
target_link_libraries(aiview_core PUBLIC aicmd_core)
add_executable(aicmd-cli src/cli/main.cpp)
target_link_libraries(aicmd-cli PRIVATE aicmd_state)
if(WIN32)
  enable_language(RC)
  add_executable(aicmd WIN32 src/gui/win32.cpp src/gui/aicmd.rc)
  target_link_libraries(aicmd PRIVATE aicmd_state comctl32 user32 gdi32 shell32 uxtheme version)
  target_link_libraries(aicmd PRIVATE aiview_core)
  target_include_directories(aicmd PRIVATE third_party)
  option(AICMD_BUILD_AIVIEW "Build optional native media viewer" ON)
  if(AICMD_BUILD_AIVIEW)
    add_library(aiview_protocol src/media/protocol.cpp)
    target_include_directories(aiview_protocol PUBLIC third_party)
    target_link_libraries(aiview_protocol PUBLIC aiview_core)
    add_executable(aiview WIN32 src/media/win32.cpp src/media/mpv_backend.cpp src/media/png.cpp src/gui/aicmd.rc)
    target_link_libraries(aiview PRIVATE aiview_protocol user32 gdi32 shell32 advapi32 ole32 windowscodecs)
    add_executable(aiview-tool src/media/tool_main.cpp)
    target_link_libraries(aiview-tool PRIVATE aiview_protocol)
  endif()
endif()
enable_testing()
add_executable(aicmd-tests tests/core_tests.cpp)
target_link_libraries(aicmd-tests PRIVATE aicmd_state)
if(WIN32)
  target_sources(aicmd-cli PRIVATE src/gui/aicmd.rc)
  target_sources(aicmd-tests PRIVATE src/gui/aicmd.rc)
endif()
add_test(NAME core COMMAND aicmd-tests)
add_executable(aicmd-ssh-tests tests/ssh_tests.cpp)
target_link_libraries(aicmd-ssh-tests PRIVATE aicmd_state)
add_test(NAME ssh-copy COMMAND aicmd-ssh-tests)
add_executable(aiview-tests tests/media_tests.cpp)
target_link_libraries(aiview-tests PRIVATE aiview_core)
add_test(NAME media COMMAND aiview-tests)
